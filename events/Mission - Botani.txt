namespace = wwu_mission_botani

# Infested Culture Spreads in Gorgrond
country_event = {
    id = wwu_mission_botani.1
    title = wwu_mission_botani.1.title
    desc = wwu_mission_botani.1.desc
    picture = BLACKROCK_DEPTHS_eventPicture

    is_triggered_only = yes
    hidden = yes
    
    trigger = {
        tag = EVB
        mission_completed = EVB_control_the_everbloom_wilds
        NOT = { mission_completed = EVB_destroy_the_foundry }
        
        any_owned_province = {
            region = region_gorgrond
            NOT = { culture_group = culture_group_botani }
        }
    }

    mean_time_to_happen = {
        days = 1
    }

    option = {
        name = wwu_mission_botani.1.option.a

        random_owned_province = {
            limit = {
                region = region_gorgrond
                NOT = { culture_group = culture_group_botani }
            }
            
            change_culture = culture_infested
        } 
    }
}
