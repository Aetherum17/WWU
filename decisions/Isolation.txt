country_decisions = {
    # Weaken Isolation
    weaken_isolation = {
        color = { 156 158 33 }
        
        potential = {
            has_country_flag = display_minor_decisions
            has_global_flag = enabled_isolation_mechanics
            can_use_isolation_mechanic = yes
            has_any_isolation = yes
        }
        
		allow = {
            NOT = { has_country_modifier = isolation_cooldown }
			stability = 1
		}
        
		effect = {
            add_stability = -1
            
            weaken_isolation_level = yes
            
            hidden_effect = {
                add_country_modifier = {
                    name = isolation_cooldown
                    duration = 3650
                    hidden = yes
                }
            }
		}
        
		ai_will_do = {
			factor = 1
            
            # AI will wait to reach 3 stability before leaving isolation
            modifier = {
                factor = 0
                
                NOT = { stability = 3 }
            }
		}
	}
    
    # Deepen Isolation
    deepen_isolation = {
        color = { 156 158 33 }
        
        potential = {
            has_country_flag = display_minor_decisions
            has_global_flag = enabled_isolation_mechanics
            can_use_isolation_mechanic = yes
            
            NOT = { has_total_isolation = yes }
        }
        
		allow = {
            NOT = { has_country_modifier = isolation_cooldown }
			stability = 1
		}
        
		effect = {
            add_stability = -1
            
            deepen_isolation_level = yes
            
            hidden_effect = {
                add_country_modifier = {
                    name = isolation_cooldown
                    duration = 3650
                    hidden = yes
                }
            }
		}
        
        # AI should not re-enter isolation
		ai_will_do = {
			factor = 0
		}
	}
